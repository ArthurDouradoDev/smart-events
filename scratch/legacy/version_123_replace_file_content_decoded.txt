            logger.warning(f"Sessão do módulo {module} expirada. Iniciando renovação automática via Playwright...")
            
            try:
                db.insert_alert({
                    "event_id":  self.event_id,
                    "level":     "GLOBAL",
                    "severity":  "WARNING",
                    "site_id":   "",
                    "cell_id":   "",
                    "message":   f"Sessão do iManager ({module}) expirada. Iniciando renovação automática via Playwright...",
                    "timestamp": datetime.utcnow().isoformat(),
                })
            except Exception as ae:
                logger.error(f"Erro ao inserir alerta de expiração no banco: {ae}")

            try:
                import subprocess
                import sys
                script_path = Path(__file__).parent.parent / "scratch" / "get_session.py"
                res = subprocess.run(
                    [sys.executable, str(script_path), "--headless"],
                    capture_output=True,
                    text=True,
                    check=True
                )
                logger.info("Playwright renovou a sessão com sucesso.")
                logger.debug(f"Saída do Playwright: {res.stdout}")
            except Exception as e:
                logger.error(f"Erro ao executar renovação de sessão: {e}")
                if hasattr(e, 'stderr') and e.stderr:
                    logger.error(f"Stderr do Playwright: {e.stderr}")
            
            self._session_monitoring = None
            self._session_trace = None
